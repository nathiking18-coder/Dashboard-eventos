Configure a sincronização com o Google Sheets
▼ Configurar
Método A — Apps Script (Recomendado)
Método B — Colar dados
Configure uma vez, sincroniza automático (a cada 10 min):
  1. Na planilha: Extensões → Apps Script
  2. Apague tudo e cole o código abaixo
  3. Implantar → Nova implantação → App da Web
  4. Executar como: Eu · Acesso: Qualquer pessoa
  5. Cole a URL gerada abaixo e clique em Conectar (vale só para este navegador)
  6. Para valer para todos que abrirem o link: cole a URL em SHEET_URL no início do script do index.html
// Apague TUDO no editor e cole este código inteiro em: Planilha → Extensões → Apps Script // Depois: Implantar → Nova implantação → App da Web // Executar como: Eu · Quem pode acessar: Qualquer pessoa function doGet(e) { var NOME_ABA = 'registro de ações comerciais'; var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = acharAba(ss, NOME_ABA); if (!sheet) return responder({ error: 'Aba "' + NOME_ABA + '" não encontrada' }, e); var tz = ss.getSpreadsheetTimeZone(); var data = sheet.getDataRange().getValues().map(function (row) { return row.map(function (v) { return Object.prototype.toString.call(v) === '[object Date]' ? Utilities.formatDate(v, tz, 'dd/MM/yyyy') : v; }); }); return responder({ data: data, atualizado: new Date().toISOString() }, e); } // Acha a aba ignorando maiúsculas, acentos e espaços extras function acharAba(ss, nome) { var alvo = normalizar(nome), abas = ss.getSheets(), i; for (i = 0; i < abas.length; i++) if (normalizar(abas[i].getName()) === alvo) return abas[i]; for (i = 0; i < abas.length; i++) if (normalizar(abas[i].getName()).indexOf('registro de acoes') === 0) return abas[i]; return null; } function normalizar(s) { return String(s).toLowerCase().normalize('NFD').replace(/[\u0300-\u036f]/g, '').replace(/\s+/g, ' ').trim(); } // JSON normal, ou JSONP quando a página pede ?callback= function responder(obj, e) { var json = JSON.stringify(obj); var cb = e && e.parameter && e.parameter.callback; if (cb && /^[\w$]+$/.test(cb)) { return ContentService.createTextOutput(cb + '(' + json + ')').setMimeType(ContentService.MimeType.JAVASCRIPT); } return ContentService.createTextOutput(json).setMimeType(ContentService.MimeType.JSON); }
